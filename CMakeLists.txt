cmake_minimum_required(VERSION 3.31.6)
project(LightYears VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
    set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "Note: including file:")
    set(CMAKE_CXX_COMPILER_LAUNCHER "${CMAKE_COMMAND};-E;env;VSLANG=1033")
    string(APPEND CMAKE_CXX_FLAGS_RELEASE " /O2 /Ob2 /Oi /Ot /GL")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_RELEASE " /LTCG")
else()
    string(APPEND CMAKE_CXX_FLAGS_RELEASE " -O3 -march=native")
endif()
set(VEGA_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Vega" CACHE PATH "Vega source checkout")
option(VEGA_BUILD_GAS "Build GAS (required by LightYears)" ON)
option(VEGA_BUILD_SKY "Build Sky (required by LightYears)" ON)
option(VEGA_BUILD_EDITOR "Build the game editor" ON)
option(VEGA_BUILD_TEMPLATES "Build engine examples" OFF)
if(NOT VEGA_BUILD_GAS OR NOT VEGA_BUILD_SKY)
    message(FATAL_ERROR "LightYears requires VEGA_BUILD_GAS and VEGA_BUILD_SKY")
endif()
add_subdirectory("${VEGA_ENGINE_DIR}" "${CMAKE_BINARY_DIR}/engine")
# Engine flags are directory-scoped. Apply the same render optimization to game visuals.
set(VEGA_DEBUG_FLAGS_RENDER "${CMAKE_CXX_FLAGS_DEBUG}")
if(MSVC AND VEGA_OPTIMIZE_RENDER_IN_DEBUG)
    string(REGEX REPLACE "/Od|/Ob[0-9]|/RTC[1csu]+" "" VEGA_DEBUG_FLAGS_RENDER "${CMAKE_CXX_FLAGS_DEBUG}")
    string(APPEND VEGA_DEBUG_FLAGS_RENDER " /O2 /Ob2")
endif()
set(LIGHT_YEARS_GAME_TARGET_NAME LightYearsGame)
add_subdirectory(LightYearsVisual)
add_subdirectory(LightYearsGame)
foreach(directory IN ITEMS "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/LightYearsVisual" "${CMAKE_CURRENT_SOURCE_DIR}/LightYearsGame")
    vega_enable_warnings_for_directory("${directory}")
endforeach()
if(BUILD_TESTING)
    add_test(NAME EngineModulesNoGameDependencies COMMAND ${CMAKE_COMMAND}
        -DVEGA_ENGINE_DIR=${VEGA_ENGINE_DIR} -DGAME_ROOT=${CMAKE_CURRENT_SOURCE_DIR}
        -P ${CMAKE_CURRENT_SOURCE_DIR}/LightYearsGame/tests/CheckSkyShaderCompatibility.cmake)
    add_test(NAME LightYearsGasLiteCore COMMAND LightYearsGasLiteTests WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME VegaTestAudioEnvironment COMMAND ${CMAKE_COMMAND}
        -DBUILD_ROOT=${CMAKE_BINARY_DIR} -P ${VEGA_ENGINE_DIR}/tools/CheckTestAudioEnvironment.cmake)
    # Must run after all game/editor registrations; covers script-launched processes too.
    vega_mute_tests("${CMAKE_CURRENT_SOURCE_DIR}")
endif()
